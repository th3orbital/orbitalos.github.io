"PM" : "AM"; hours = hours % 12 || 12; const timeStr = `${hours}:${minutes}`; const timeEl = document.getElementById("time"); const navTime = document.getElementById("nav-time"); const dateEl = document.getElementById("date"); if (timeEl) timeEl.textContent = timeStr; if (navTime) navTime.textContent = `${timeStr} ${ampm}`; if (dateEl) { dateEl.textContent = now.toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric" }).toUpperCase(); } } setInterval(updateClock, 1000); updateClock();
